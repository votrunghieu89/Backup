-- 1. Create table DanhMucHoa (Category)
Create table DanhMucHoa(
	MaDM int identity(1,1) primary key,
	TenDM nvarchar(255) not null,
	MoTa nvarchar(max) -- Đã sửa MaTa thành MoTa theo ERD
)

---

-- 2. Create table Hoa (Product)
Create table Hoa (
	MaHoa int identity(1,1) primary key,
	TenHoa nvarchar(255) not null,
	DonViTinh varchar(20),
	DonGia decimal(18,2), -- Đã tăng kích thước để chứa giá trị lớn hơn
	HinhAnh varchar(max),
	MoTa nvarchar(max),
	MaDM int,
	Foreign key (MaDM) references DanhMucHoa(MaDM)
)

---

-- 3. Create table KhachHang (Customer)
Create table KhachHang(
	TenDN varchar(255) primary key,
	MatKhau varchar(255),
	TenKhachHang nvarchar(255)
)

---

-- 4. Create table DonHang (Order Details)
Create table DonHang(
	TenDN varchar(255),
	MaHoa int,
	SoLuong int,
	primary key(TenDN, MaHoa),
	Foreign key (TenDN) references KhachHang(TenDN),
	Foreign key (MaHoa) references Hoa(MaHoa) -- Đã sửa: Phải tham chiếu đến bảng Hoa
)
